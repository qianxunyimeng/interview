
## async和await

 - async函数返回的 Promise 对象，必须等到内部所有await命令后面的 Promise 对象执行完，才会发生状态改变，除非遇到return语句或者抛出错误。也就是说，只有async函数内部的异步操作执行完，才会执行then方法指定的回调函数。


## 最大并发任务(请求)

如果有100个请求，如何控制并发？

```js
// 请求列表,这里用setTimeout模拟异步任务
const requestList = [];
for (let i = 1; i <= 100; i++) {
  requestList.push(
    () =>
      new Promise((resolve, reject) => {
        setTimeout(() => {
          if (i === 92) {
            reject(new Error(`请求异常：【${i}】`));
          } else {
            console.log('done', i);
            resolve(`请求完成【${i}】`);
          }
        }, Math.random() * 1000);
      }),
  );
}

/**
 * 
 * @param {*} urls 请求url数组
 * @param {*} maxNum 最大并发请求数
 */
function concurRequest (urls, maxNum) {
  if (urls.length == 0) {
    return Promise.resolve([])
  }

  return new Promise((resolve, reject) => { 
    let nextIndex = 0; // 下一个请求索引
    const result = []; // 放置请求结果
    let finished = 0;// 请求完成的数量
    async function _request () { 
      if (nextIndex >= urls.length) {// 所有请求都发出
        return
      }
      const i = nextIndex
      const url = urls[nextIndex++]
      try {
        const resp = await url()
        result[i] = resp
      } catch (error) {
        
      } finally {
        finished++
        if (finished == urls.length) {
          resolve(result)
        }
      }
      _request() // 继续下一个请求
    }

    // 第一轮并发任务，相当于是启动任务
    for (let i = 0; i < Math.min(maxNum, urls.length); i++) {
      await _request()
    }
  })
}

async function main () { 
  const res = await concurRequest(requestList, 10)
  console.log(res)
}

main()
```

```js

function pLimit(concurrency) {
  validateConcurrency(concurrency);
  const queue = [];
  // // 当前正在执行的任务数量
  let activeCount = 0;

  const resumeNext = () => {
    // 如果队列中还有任务，则出队并执行
    if (activeCount < concurrency && queue.length > 0) {
      queue.shift()();
      activeCount++;
    }
  };

  const next = () => {
    activeCount--;
    resumeNext();
  };

  const run = async (function_, resolve, arguments_) => {
    // 执行任务函数fn，并将结果保存到result中
    const result = (async () => function_(...arguments_))();

    resolve(result);

    try {
      await result;
    } catch {}
  // 执行完任务后调用next函数，继续执行下一个任务
    next();
  };

// 入队任务的函数
  const enqueue = (function_, resolve, arguments_) => {
    new Promise(internalResolve => {
      queue.push(internalResolve);
    }).then(
      run.bind(undefined, function_, resolve, arguments_),
    );

    (async () => {
       // 这个函数需要在下一个微任务中执行，以便在比较`activeCount`和`concurrency`之前等待
      // `activeCount`在任务函数出队并调用时会异步更新。if语句中的比较也需要异步执行，以获取`activeCount`的最新值。
      await Promise.resolve();
      // 如果当前正在执行的任务数量小于并发数并且队列中还有任务，则出队并执行
      if (activeCount < concurrency) {
        resumeNext();
      }
    })();
  };

  const generator = (function_, ...arguments_) => new Promise(resolve => {
    enqueue(function_, resolve, arguments_);
  });
  return generator;
}
// 检查并发数concurrency是否为正整数或正无穷大，并且大于0
function validateConcurrency(concurrency) {
  if (!((Number.isInteger(concurrency) || concurrency === Number.POSITIVE_INFINITY) && concurrency > 0)) {
    throw new TypeError('Expected `concurrency` to be a number from 1 and up');
  }
}

}
```
