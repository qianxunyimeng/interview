

## 用递归算法实现，数组长度为5且元素的随机数在2 - 32间不重复的值

```js
// 指定区间[n,m]的随机数
function getRandomNum (min, max) {
  return Math.floor(Math.random() * (max + 1 - min) + min)
}

// 得到一个随机整数数组
function getUniqueArr (arr = [],len = 1,min = 0,max = 1) { 
  if (arr.length < len) {
    const randomNum = getRandomNum(min,max)
    if (arr.indexOf(randomNum) < 0) {
      arr.push(randomNum)
    }
    getUniqueArr(arr,len,min,max)
  }
  return arr
}
console.log(getUniqueArr([],5,2,32));
```

## 写一个方法去掉字符串中的空格

```js
function trim (str, type = "") { 
  let Reg = '';
  switch (type) {
    case "left":
      Reg = /^\s*/
      break;
    case "right":
      Reg = /\s*$/
      break;  
    case "both": // 去除两端的空格
      Reg = /^\s*|\s*$/g
      break
    default: // 默认去除所有空格
      Reg = /\s*/g
      break;
  }
  let newStr = str.replace(Reg, '');
  if (type == 'middle') {
    let RegLeft = str.match(/(^\s*)/g)[0]; // 保存左边开头的空格
    let RegRight = str.match(/(\s*$)/g)[0]; // 保存右边尾端的空格
    newStr = RegLeft + newStr + RegRight; // 将空格加给清完全部空格后的字符串
  }
  return newStr;
}

const str = "  我爱你 中国  ! "
console.log(`【去除所有空格】 原字符串: ${str},len:${str.length}, 处理后: ${trim(str)},len:${trim(str).length}`)

console.log(`【去除右部空格】 原字符串: ${str},len:${str.length}, 处理后: ${trim(str, 'right')},len:${trim(str, 'right').length}`)
console.log(`【去除前后空格】 原字符串: ${str},len:${str.length}, 处理后: ${trim(str, 'both')},len:${trim(str, 'both').length}`)
console.log(`【去除中间空格】 原字符串: ${str},len:${str.length}, 处理后: ${trim(str, 'middle')},len:${trim(str, 'middle').length}`)
```

## 去除字符串中最后一个指定的字符(字符串)

```js
// 正则
function delLast(str,target) {
  let reg =new RegExp(`${target}(?=([^${target}]*)$)`)
  return str.replace(reg,'')
}
console.log(delLast('abcdef', 'b')); // acdef
console.log(delLast('abcdefabcd', 'a')); // abcdefbcd
console.log(delLast('abcdefabcd', 'abc')); // abcdefd
console.log(delLast('好好学习,天天向上,学习使我快乐', '学习'));// 好好学习,天天向上,使我快乐
```

## 下划线命名转驼峰命名

一般使用小驼峰命名法

```js
/**
 * 下划线命名转大驼峰命名
    UnderScoreToCamel('sql_fun_input') ==> SqlFunInput
 *
 */
function UnderScoreToUpperCamel (name) { 
  return name.replace(/(^|_)(\w)/g, (m, $1, $2) => {
    return $2.toUpperCase()
  });
}
console.log(UnderScoreToUpperCamel('sql_fun_input')); //SqlFunInput

/**
 * 下划线命名转小驼峰命名
 */
function UnderScoreToLowerCamel (name) { 
  return name.replace(/_(\w)/g, (m, $1) => {
    return $1.toUpperCase()
  });
}
console.log(UnderScoreToLowerCamel('sql_fun_input')); //sqlFunInput
```

## 驼峰命名转下划线命名

```js
function toUnderline (str) {
  return str.replace(/([A-Z])/g, '_$1').toLowerCase();
}

console.log(toUnderline('systemUsername')); // 输出 'system_username'
console.log(toUnderline('myCode')); // 输出 'my_code'
```

## 大写转小写，小写转大写

```js
function caseConvert(str){
    return str.replace(/([a-z]*)([A-Z]*)/g, (m, s1, s2)=>{
     return `${s1.toUpperCase()}${s2.toLowerCase()}`
    })
}
```
